async function fetchStudentData(studentId) {
    try {
        // 'await' pauses function execution until promise resolves
        const response = await fetch(`https://api.university.edu/students/${studentId}`);

        console.log("Student Record:", data);

        console.error("Network or API failure:", error.message);
    }

8. React Hooks: `useState` & `useEffect` Dependency Array ⭐⭐⭐⭐⭐

Hooks let functional components use state and lifecycle side-effects without writing legacy ES6 classes.

Controlled Form & useEffect Dependency Variations
import React, { useState, useEffect } from "react";

function StudentDashboard({ courseId }) {
    // 1. useState: Manages local component state
    const [students, setStudents] = useState([]);
    const [searchTerm, setSearchTerm] = useState("");

    // Case 1: Empty Dependency Array [] -> Runs EXACTLY ONCE after initial component mount
    useEffect(() => {
        console.log("Component mounted: fetching initial data...");
    }, []);

    // Case 2: Specific Dependency Array [courseId] -> Runs whenever 'courseId' prop changes
    useEffect(() => {
        fetch(`/api/courses/${courseId}/students`)
            .then(res => res.json())
            .then(data => setStudents(data));
    }, [courseId]);

    // Case 3: No Dependency Array -> Runs after EVERY single render (use with caution!)
    useEffect(() => {
        document.title = `Students (${students.length})`;
    });

    return (
        <div>
            <input 
                type="text" 
                value={searchTerm} 
                onChange={(e) => setSearchTerm(e.target.value)} 
                placeholder="Search students..."
            />
            <p>Active Filter: {searchTerm}</p>
        </div>
    );
}

Why non-blocking? Node does not sit idle waiting for I/O operations (file reading, DB queries, network calls). It delegates them to the system/thread pool and continues executing other code:

Async Demonstration (fs.readFile)

function checkAuthorization(req, res, next) {
    const authHeader = req.headers.authorization;

    // Check 1: Existence of Authorization header
    if (!authHeader) {
        return res.status(401).json({
            error: "Unauthorized",
            message: "Authorization header is missing"
        });
    }

    // Check 2: Correct RFC 6750 Bearer token format
    if (!authHeader.startsWith("Bearer ")) {
        return res.status(401).json({
            error: "Unauthorized",
            message: "Invalid authorization format. Must be 'Bearer <token>'"
        });
    }

    const token = authHeader.split(" ")[1];
    if (!token || token.trim() === "") {
        return res.status(401).json({ error: "Unauthorized", message: "Token payload empty" });
    }

    // Pass control to the protected route handler
    next();
}

Q13. Request Processing vs Response Generation

Lifecycle Concept

Every web interaction splits cleanly into two distinct phases on the server:

  1. Request Processing (Inbound Phase):
    • Receives incoming raw TCP/HTTP stream.
    • Parses URL path parameters (req.params) and search query string (req.query).
    • Parses incoming request JSON payload (req.body).
    • Executes authentication middleware and input validation.
    • Runs database query or business rule computations.
  2. Response Generation (Outbound Phase):
    • Takes the business logic / database result.
    • Serializes data into standard format (typically JSON or HTML view).
    • Sets appropriate HTTP status header (e.g., 200 OK, 201 Created, 404 Not Found).
    • Sets Content-Type: application/json header and streams data back to client socket.

Q14. Database Connection Pools (High-Concurrency Optimization)

Critical Architecture Concept

The Problem with Single/Ad-hoc Connections: If 100 students submit requests simultaneously, establishing a new TCP handshake, authenticating, and tearing down 100 separate database connections introduces enormous latency and server memory exhaustion.

The Connection Pool Solution: A Connection Pool maintains a warm, reusable cache of persistent database connections (e.g., connectionLimit: 10). Requests grab an available connection from the pool, execute their query, and immediately release it back to the pool!
Configuring Connection Pool with mysql2
const mysql = require("mysql2");

// Create Connection Pool
const pool = mysql.createPool({
    host: "localhost",
    user: "root",
    password: process.env.DB_PASSWORD || "",
    database: "college",
    connectionLimit: 10,       // Max 10 simultaneous connections
    waitForConnections: true,  // Queue requests when all 10 connections are busy
    queueLimit: 0              // Unlimited request queue
});

// Queries automatically borrow, execute, and release connections:
pool.query("SELECT * FROM students", (err, results) => {
    if (err) throw err;
    console.log("Students retrieved via pool:", results);

function authenticateToken(req, res, next) {
    const authHeader = req.headers["authorization"];
    const token = authHeader && authHeader.split(" ")[1]; // Bearer <TOKEN>

        next();

            throw new Error(errBody.error || `HTTP error! status: ${response.status}`);
        }

        const data = await response.json();
        setStudents(data);
    } catch (err) {
        setError(err.message);
    } finally {
        setLoading(false);
    }
}

const express = require("express");
const router = express.Router();
const db = require("./db"); // mysql2/promise connection pool
const { authenticateToken } = require("./authMiddleware");

// 1. READ ALL STUDENTS
router.get("/", authenticateToken, async (req, res, next) => {
    try {
        const [rows] = await db.query("SELECT id, name, email, course, gpa FROM students ORDER BY id DESC");
        res.status(200).json(rows);
    } catch (err) { next(err); }
});

// 2. CREATE NEW STUDENT
router.post("/", authenticateToken, async (req, res, next) => {
    try {
        const { name, email, course, gpa } = req.body;
        if (!name || !email) return res.status(400).json({ error: "Name and email are required" });

        const [result] = await db.query(
            "INSERT INTO students (name, email, course, gpa) VALUES (?, ?, ?, ?)",
            [name, email, course, gpa]
        );
        res.status(201).json({ message: "Student enrolled", studentId: result.insertId });
    } catch (err) { next(err); }
});

// 3. UPDATE EXISTING STUDENT
router.put("/:id", authenticateToken, async (req, res, next) => {
    try {
        const { id } = req.params;
        const { name, email, course, gpa } = req.body;
        const [result] = await db.query(
            "UPDATE students SET name = ?, email = ?, course = ?, gpa = ? WHERE id = ?",
            [name, email, course, gpa, id]
        );
        if (result.affectedRows === 0) return res.status(404).json({ error: "Student not found" });
        res.status(200).json({ message: "Student updated successfully" });
    } catch (err) { next(err); }
});

// 4. DELETE STUDENT
router.delete("/:id", authenticateToken, async (req, res, next) => {
    try {
        const { id } = req.params;
        const [result] = await db.query("DELETE FROM students WHERE id = ?", [id]);
        if (result.affectedRows === 0) return res.status(404).json({ error: "Student not found" });
        res.status(200).json({ message: "Student deleted successfully" });
    } catch (err) { next(err); }
});

  1. Node Event Loop: Single-threaded JavaScript handles non-blocking I/O via libuv worker threads and callback queues.
